Exercise 23: Black-Scholes call price (formula)



Difficulty: Advanced.



Show solution



  

    
RInteractive R

    
bs_call <- function(S, K, r, T, sigma) {
  d1 <- (log(S/K) + (r + 0.5*sigma^2)*T) / (sigma*sqrt(T))
  d2 <- d1 - sigma*sqrt(T)
  S*pnorm(d1) - K*exp(-r*T)*pnorm(d2)
}
bs_call(S = 100, K = 100, r = 0.05, T = 1, sigma = 0.2)
