# Average mpg by cylinders, plus the spread
mtcars |>
  group_by(cyl) |>
  summarise(
    n = n(),
    avg = mean(mpg),
    sd = sd(mpg)
  ) |>
  arrange(desc(avg))

Exercise 2.3 Top-revenue regions, sorted

Task. Group sales by region and return mean revenue, sorted highest first.

# your answer
ex_2_3 <- sales |>
  group_by(region) |>
  summarise(avg = mean(revenue)) |>
  arrange(desc(avg))
