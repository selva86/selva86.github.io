RMonthly rainfall anomalies, Chennai
# 30 years of Chennai rainfall vs the seasonal norm
library(ggplot2)

rain |>
  mutate(anomaly = mm - norm) |>
  ggplot(aes(month, anomaly, fill = anomaly > 0)) +
  geom_col(show.legend = FALSE) +
  scale_fill_manual(values = c("#b3541e", "#2056d2")) +
  labs(x = NULL, y = "mm vs norm") +
  theme_minimal()
